Scenario 1 (15 lines):

WRITE “Which restriction?” (egg-free)
READ chosen_label
WRITE “Maximum preparation time in minutes?” (20 minutes)
READ max_time

found = 0
READ first recipe (recipe_ID, recipe_name, prep_time, dietary_labels)

WHILE recipe meets requirements
    IF dietary_labels contains chosen_label
        IF prep_time < max_time
            WRITE recipe_ID, recipe_name, prep_time, dietary_labels
            found = found + 1
        ENDIF
    ENDIF
    READ next recipe
ENDWHILE

IF found = 0
    WRITE "No recipes were found."
ENDIF
